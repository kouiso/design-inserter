🎭 Playwright テストコード例

テストコードを表示 
import { test, expect } from '@playwright/test';

const BASE_URL = 'http://localhost:10010';

// 固定ページ一覧
const pages = [
  { name: 'TOP', url: '/' },
  { name: '製品情報', url: '/product/' },
  { name: '注目製品', url: '/featured/' },
  // ... 省略
];

// 1. ページ表示テスト
test.describe('ページ表示テスト', () => {
  for (const page of pages) {
    test(`${page.name}ページが正常に表示される`, async ({ page: browserPage }) => {
      const response = await browserPage.goto(`${BASE_URL}${page.url}`);
      expect(response?.status()).toBe(200);
      await expect(browserPage.locator('header')).toBeVisible();
      await expect(browserPage.locator('footer')).toBeVisible();
    });
  }
});

// 3. ページネーションテスト
test.describe('ページネーションテスト', () => {

  test('製品情報ページのページネーション', async ({ page }) => {
    await page.goto(`${BASE_URL}/product/`);

    // ページ2へ遷移
    await page.locator('.pagination a:has-text("2")').click();
    await expect(page).toHaveURL(/\/product\/page\/2\//);

    // ページ3へ遷移
    await page.locator('.pagination a:has-text("3")').click();
    await expect(page).toHaveURL(/\/product\/page\/3\//);
  });

  test('インタビューページのページネーション', async ({ page }) => {
    await page.goto(`${BASE_URL}/career/interview/`);

    await page.locator('.pagination a:has-text("2")').click();
    await expect(page).toHaveURL(/\/career\/interview\/page\/2\//);
  });

  test('ストーリーページのページネーション', async ({ page }) => {
    await page.goto(`${BASE_URL}/story/`);

    await page.locator('.pagination a:has-text("2")').click();
    await expect(page).toHaveURL(/\/story\/page\/2\//);
  });
});
